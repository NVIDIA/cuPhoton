cmake_minimum_required(VERSION 3.24)
project(xbeam_native LANGUAGES CXX)
add_library(xbeam_radio STATIC src/vdif.cpp src/swin.cpp src/requests.cpp src/job_text.cpp src/job.cpp src/index.cpp src/timeline.cpp src/replay.cpp src/window.cpp src/recording.cpp src/execution_bridge.cpp)
target_include_directories(xbeam_radio PUBLIC include PRIVATE third_party)
target_compile_features(xbeam_radio PUBLIC cxx_std_17)
set_target_properties(xbeam_radio PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(xbeam_radio PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
target_link_libraries(xbeam_radio PUBLIC OpenSSL::Crypto)
add_executable(xbeam-native src/main.cpp)
target_link_libraries(xbeam-native PRIVATE xbeam_radio)
install(TARGETS xbeam_radio xbeam-native)
install(DIRECTORY include/ DESTINATION include)

option(XBEAM_ENABLE_CUDA "Build native CUDA execution" OFF)
if(XBEAM_ENABLE_CUDA)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    if(CUDAToolkit_VERSION VERSION_LESS 13 OR NOT CUDAToolkit_VERSION VERSION_LESS 14)
        message(FATAL_ERROR "The native xBeam backend requires a CUDA 13 toolkit")
    endif()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(generated "${CMAKE_CURRENT_BINARY_DIR}/kernels")
    add_custom_command(OUTPUT "${generated}/front.inc" "${generated}/channel.inc" "${generated}/products.inc" "${generated}/multi_products.inc"
        COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/cuda/generate.py"
                "${CMAKE_CURRENT_SOURCE_DIR}/.." "${generated}"
        DEPENDS cuda/generate.py ../cuda_front.py ../cuda_channelize.py ../core/cuda_products.py)
    add_library(xbeam_cuda STATIC cuda/front.cu cuda/channel.cu cuda/products.cu cuda/multi_products.cu cuda/station.cpp cuda/input.cu cuda/input.cpp cuda/series.cpp cuda/series_kernels.cu
                "${generated}/front.inc" "${generated}/channel.inc" "${generated}/products.inc" "${generated}/multi_products.inc")
    target_include_directories(xbeam_cuda PUBLIC include PRIVATE "${generated}" third_party)
    target_compile_features(xbeam_cuda PUBLIC cxx_std_17)
    set_target_properties(xbeam_cuda PROPERTIES CUDA_STANDARD 17 POSITION_INDEPENDENT_CODE ON)
    target_link_libraries(xbeam_cuda PUBLIC xbeam_radio CUDA::cudart CUDA::cufft)
endif()
if(XBEAM_ENABLE_CUDA)
    add_executable(xbeam-native-station-test cuda/station_test.cpp)
    target_link_libraries(xbeam-native-station-test PRIVATE xbeam_cuda)
endif()
if(XBEAM_ENABLE_CUDA)
    add_executable(xbeam-native-station-file-test cuda/station_file_test.cpp)
    target_include_directories(xbeam-native-station-file-test PRIVATE third_party)
    target_link_libraries(xbeam-native-station-file-test PRIVATE xbeam_cuda)
endif()
if(XBEAM_ENABLE_CUDA)
    add_executable(xbeam-native-products-test cuda/products_test.cpp)
    target_link_libraries(xbeam-native-products-test PRIVATE xbeam_cuda)
endif()
if(XBEAM_ENABLE_CUDA)
    add_library(xbeam_execution STATIC cuda/radio_run.cpp cuda/radio_options.cpp cuda/radio_model.cpp)
    target_include_directories(xbeam_execution PRIVATE third_party)
    target_compile_options(xbeam_execution PRIVATE -ffp-contract=off)
    set_target_properties(xbeam_execution PROPERTIES POSITION_INDEPENDENT_CODE ON)
    find_package(Threads REQUIRED)
    target_link_libraries(xbeam_execution PUBLIC xbeam_cuda Threads::Threads)
    add_executable(xbeam-native-correlate cuda/main.cpp)
    target_link_libraries(xbeam-native-correlate PRIVATE xbeam_execution)
    install(TARGETS xbeam_execution xbeam_cuda xbeam-native-correlate)
endif()

option(XBEAM_ENABLE_CUFILE "Enable native cuFile reads" OFF)
if(XBEAM_ENABLE_CUDA AND XBEAM_ENABLE_CUFILE)
    find_library(XBEAM_CUFILE_LIBRARY cufile HINTS "${CUDAToolkit_LIBRARY_DIR}" "${CUDAToolkit_ROOT}/lib64" REQUIRED)
    target_compile_definitions(xbeam_cuda PRIVATE XBEAM_HAVE_CUFILE=1)
    target_link_libraries(xbeam_cuda PUBLIC "${XBEAM_CUFILE_LIBRARY}")
endif()
if(XBEAM_ENABLE_CUDA)
    add_executable(xbeam-native-input-test cuda/input_test.cpp)
    target_link_libraries(xbeam-native-input-test PRIVATE xbeam_cuda Threads::Threads)
endif()
